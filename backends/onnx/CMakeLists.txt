cmake_minimum_required(VERSION 3.16)
project(pron_onnx CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(PRON_CORE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../core/include" CACHE PATH "pron core headers")
set(ONNXRUNTIME_VERSION "1.22.0" CACHE STRING "onnxruntime release to download")
set(ONNXRUNTIME_ROOT "" CACHE PATH "Prebuilt onnxruntime (include/ + lib/); downloaded if empty")

if(NOT ONNXRUNTIME_ROOT)
  include(FetchContent)
  set(_ort_name onnxruntime-linux-x64-${ONNXRUNTIME_VERSION})
  if(WIN32)
    set(_ort_name onnxruntime-win-x64-${ONNXRUNTIME_VERSION})
    set(_ort_ext zip)
  else()
    set(_ort_ext tgz)
  endif()
  FetchContent_Declare(onnxruntime_prebuilt
    URL https://github.com/microsoft/onnxruntime/releases/download/v${ONNXRUNTIME_VERSION}/${_ort_name}.${_ort_ext})
  FetchContent_MakeAvailable(onnxruntime_prebuilt)
  set(ONNXRUNTIME_ROOT "${onnxruntime_prebuilt_SOURCE_DIR}")
endif()

find_library(ORT_LIB NAMES onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" "${ONNXRUNTIME_ROOT}/jni/${ANDROID_ABI}" NO_DEFAULT_PATH)
if(NOT ORT_LIB)
  message(FATAL_ERROR "onnxruntime library not found under ${ONNXRUNTIME_ROOT}")
endif()
find_path(ORT_INCLUDE_DIR onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" "${ONNXRUNTIME_ROOT}/headers" NO_DEFAULT_PATH)
if(NOT ORT_INCLUDE_DIR)
  message(FATAL_ERROR "onnxruntime headers not found under ${ONNXRUNTIME_ROOT}")
endif()

add_library(pron_onnx STATIC src/silero_vad.cpp src/wav2vec2_phoneme.cpp src/vocab_json.cpp)
target_include_directories(pron_onnx PUBLIC include "${PRON_CORE_INCLUDE_DIR}" PRIVATE "${ORT_INCLUDE_DIR}" src)
target_link_libraries(pron_onnx PUBLIC "${ORT_LIB}")
set_target_properties(pron_onnx PROPERTIES POSITION_INDEPENDENT_CODE ON)

option(PRON_ONNX_BUILD_TESTS "Build pron_onnx tests" ON)
if(PRON_ONNX_BUILD_TESTS)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(_dl "${CMAKE_BINARY_DIR}/test_data")
  file(MAKE_DIRECTORY "${_dl}")
  set(_silero "${_dl}/silero_vad.onnx")
  if(NOT EXISTS "${_silero}")
    file(DOWNLOAD "https://raw.githubusercontent.com/snakers4/silero-vad/v6.2/src/silero_vad/data/silero_vad.onnx"
         "${_silero}" STATUS _st TLS_VERIFY ON)
    list(GET _st 0 _code)
    if(NOT _code EQUAL 0)
      file(REMOVE "${_silero}")
      message(WARNING "Silero download failed: ${_st}")
    endif()
  endif()
  add_executable(pron_onnx_tests tests/test_onnx.cpp)
  target_link_libraries(pron_onnx_tests PRIVATE pron_onnx)
  target_compile_definitions(pron_onnx_tests PRIVATE
    SILERO_MODEL="${_silero}" FAKE_MODEL="${_dl}/fake_w2v.onnx" FAKE_VOCAB="${_dl}/fake_vocab.json")
  if(NOT WIN32)
    set_target_properties(pron_onnx_tests PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
  endif()
  add_custom_command(TARGET pron_onnx_tests PRE_BUILD
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fake_model.py" "${_dl}"
    VERBATIM)
  add_test(NAME pron_onnx_tests COMMAND pron_onnx_tests)
endif()
