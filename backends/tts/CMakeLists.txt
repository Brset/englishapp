cmake_minimum_required(VERSION 3.16)
project(pron_tts CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(PRON_CORE_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../core/include" CACHE PATH "pron core headers")
set(SHERPA_ONNX_VERSION "1.12.14" CACHE STRING "sherpa-onnx release")
set(SHERPA_ONNX_ROOT "" CACHE PATH "Unpacked sherpa-onnx shared release (include/, lib/)")
set(PRON_TTS_VOICE_URL "https://github.com/k2-fsa/sherpa-onnx/releases/download/tts-models/vits-piper-en_US-lessac-medium.tar.bz2" CACHE STRING "test voice")

if(NOT SHERPA_ONNX_ROOT)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT ANDROID)
    set(_n "sherpa-onnx-v${SHERPA_ONNX_VERSION}-linux-x64-shared")
    set(_d "${CMAKE_BINARY_DIR}/_deps/${_n}")
    if(NOT EXISTS "${_d}/${_n}/lib")
      file(MAKE_DIRECTORY "${_d}")
      set(_f "${_d}/${_n}.tar.bz2")
      message(STATUS "Downloading ${_n}")
      file(DOWNLOAD "https://github.com/k2-fsa/sherpa-onnx/releases/download/v${SHERPA_ONNX_VERSION}/${_n}.tar.bz2" "${_f}" STATUS _st TLS_VERIFY ON)
      list(GET _st 0 _rc)
      if(NOT _rc EQUAL 0)
        message(FATAL_ERROR "sherpa-onnx download failed: ${_st}")
      endif()
      file(ARCHIVE_EXTRACT INPUT "${_f}" DESTINATION "${_d}")
    endif()
    set(SHERPA_ONNX_ROOT "${_d}/${_n}")
  else()
    message(FATAL_ERROR "Set -DSHERPA_ONNX_ROOT=<unpacked sherpa-onnx shared release> (see README.md)")
  endif()
endif()

find_library(SHERPA_C_API NAMES sherpa-onnx-c-api sherpa-onnx-jni HINTS "${SHERPA_ONNX_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(SHERPA_ORT NAMES onnxruntime HINTS "${SHERPA_ONNX_ROOT}/lib" NO_DEFAULT_PATH)

add_library(pron_tts STATIC src/piper_tts.cpp)
target_include_directories(pron_tts PUBLIC include "${PRON_CORE_INCLUDE_DIR}" PRIVATE "${SHERPA_ONNX_ROOT}/include")
target_link_libraries(pron_tts PUBLIC "${SHERPA_C_API}")
if(SHERPA_ORT)
  target_link_libraries(pron_tts PUBLIC "${SHERPA_ORT}")
endif()
set_target_properties(pron_tts PROPERTIES BUILD_RPATH "${SHERPA_ONNX_ROOT}/lib")

enable_testing()
add_executable(pron_tts_test test/tts_test.cpp)
target_link_libraries(pron_tts_test PRIVATE pron_tts)
add_test(NAME tts_fetch_voice COMMAND ${CMAKE_COMMAND}
  -DURL=${PRON_TTS_VOICE_URL} -DDEST=${CMAKE_BINARY_DIR}/voice -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/fetch_voice.cmake)
add_test(NAME tts_synthesize COMMAND pron_tts_test ${CMAKE_BINARY_DIR}/voice ${CMAKE_BINARY_DIR})
set_tests_properties(tts_synthesize PROPERTIES DEPENDS tts_fetch_voice SKIP_RETURN_CODE 77)
